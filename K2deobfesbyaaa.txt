local function buildConfig()

        AutoWriteEnabled      = _G.AutoWriteEnabled == true,
        AutoSubmitEnabled     = _G.AutoSubmitEnabled == true,

        SubmitDelay           = tonumber(_G.SubmitDelay) or 0,
        CasingType            = tostring(_G.CasingType or "Upper"),
        TypeSpeed             = tostring(_G.TypeSpeed or "Fast"),

local function applyConfig(cfg)
    if type(cfg) ~= "table" then return end
    if cfg.AutoWriteEnabled ~= nil then _G.AutoWriteEnabled = cfg.AutoWriteEnabled == true end
    if cfg.AutoSubmitEnabled ~= nil then _G.AutoSubmitEnabled = cfg.AutoSubmitEnabled == true end
    if cfg.RiddleSolverEnabled ~= nil then _G.RiddleSolverEnabled = cfg.RiddleSolverEnabled == true end
    if cfg.AIRiddleEnabled ~= nil then _G.AIRiddleEnabled = cfg.AIRiddleEnabled == true end
    if cfg.SammyFilterEnabled ~= nil then _G.SammyFilterEnabled = cfg.SammyFilterEnabled ~= false end
    if cfg.SubmitAfterCount ~= nil then
        _G.SubmitAfterCount = math.clamp(math.floor(tonumber(cfg.SubmitAfterCount) or 1), 1, 20)
    end
    if cfg.SubmitDelay ~= nil then
        _G.SubmitDelay = math.clamp(tonumber(cfg.SubmitDelay) or 0, 0, 2)
    end
    if cfg.CasingType ~= nil then
        local v = tostring(cfg.CasingType)
        if v == "Upper" or v == "Lower" or v == "Normal" then _G.CasingType = v end
    end
    if cfg.TypeSpeed ~= nil then
        local v = tostring(cfg.TypeSpeed)
        if v == "Fast" or v == "Normal" then _G.TypeSpeed = v end
    end
    if cfg.TrackersEnabled ~= nil then _G.TrackersEnabled = cfg.TrackersEnabled ~= false end
    if cfg.AnchorEnabled ~= nil then _G.AnchorEnabled = cfg.AnchorEnabled == true end
    if cfg.AutoBuyEnabled ~= nil then _G.AutoBuyEnabled = cfg.AutoBuyEnabled == true end
    if cfg.bgIndex ~= nil then
        _G.K2SniperBgIndex = math.clamp(math.floor(tonumber(cfg.bgIndex) or 1), 1, 9)
    end
end

    local ok = pcall(function()
        local wf = writefile or (syn and syn.writefile)
        if not wf then return end
        wf(CONFIG_FILE, HttpService:JSONEncode(buildConfig()))
    end)
    return ok
end

local function loadConfig()
    local ok, data = pcall(function()
        local rf = readfile or (syn and syn.readfile)
        local isf = isfile or (syn and syn.isfile)
        if not rf then return nil end
        if isf and not isf(CONFIG_FILE) then return nil end
        local content = rf(CONFIG_FILE)
        if type(content) ~= "string" or #content < 2 then return nil end
        return HttpService:JSONDecode(content)
    end)
    if ok and type(data) == "table" then
        applyConfig(data)
        return true
    end
    return false
end

        savingConfig = false
        saveConfig()
    end)
end

local sortedKeys = nil
local function getSortedKeys()
    if sortedKeys then return sortedKeys end
    sortedKeys = {}
    for k in pairs(riddleDictionary) do table.insert(sortedKeys, k) end
    table.sort(sortedKeys, function(a, b) return #a > #b end)
    return sortedKeys
end

local function solveRiddleWithDictionary(text)
    local clean = tostring(text or ""):lower():gsub("[^%w]", "")
    if clean == "" then return nil end

    local parts = {}
    local remaining = clean
    local guard = 0
    local keys = getSortedKeys()

    while #remaining > 0 and guard < 30 do
        guard += 1
        local bestPos, bestEnd, bestKey = nil, nil, nil
        for _, key in ipairs(keys) do
            local k = key:lower():gsub("[^%w]", "")
            if #k >= 2 then
                local pos, endPos = remaining:find(k, 1, true)
                if pos and pos <= 10 then
                    if not bestPos or pos < bestPos or (pos == bestPos and #k > #(bestKey or "")) then
                        bestPos, bestEnd, bestKey = pos, endPos, key
                    end
                end
            end
        end
        if not bestKey then
            remaining = remaining:sub(2)
        else
            local value = riddleDictionary[bestKey]
            local rest = remaining:sub(bestEnd + 1)
            local multiplier = 1
            if rest:sub(1, 5) == "twice" then
                multiplier = 2
                rest = rest:sub(6)
            else
                local num = rest:match("^(%d+)times")
                if num then
                    multiplier = tonumber(num) or 1
                    rest = rest:gsub("^%d+times", "", 1)
                end
            end
            table.insert(parts, string.rep(value, multiplier))
            remaining = rest
        end
    end
    if #parts == 0 then return nil end
    return table.concat(parts, "")
end
